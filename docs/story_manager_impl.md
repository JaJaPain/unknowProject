## SECTION 1 — GlobalState Additions

Add these variables to `scripts/GlobalState.gd` alongside the existing
`kaelen_briefing_seen` and `kaelen_arrival_systems_seen` declarations.

```gdscript
# ── Story Manager state ───────────────────────────────────────────────────────
var story_arc: Dictionary = {}          # generated by Gemma4 on first campaign load
var story_state: Dictionary = {         # runtime + persisted progress
    "fired_beats": {},                  # beat_id (String) → Unix timestamp (int)
    "arc_flags": {},                    # arc_id (String) → value (String)
    "current_arc_index": 0,            # which arc in story_arc["arcs"] is active
    "beats_since_progress": 0,         # stall counter
    "active_nudge_level": 0,           # 0 = none, 1–5 = escalating pressure
}
var story_quest_hint: Dictionary = {}   # written by StoryManager, read by LLMInterface
var story_world_pressure: Dictionary = {} # written by StoryManager, read by MainScene
```

### Save additions

In GlobalState's `get_save_data()` (or equivalent), add:

```gdscript
data["story_arc"]   = story_arc
data["story_state"] = story_state
```

Do NOT save `story_quest_hint` or `story_world_pressure` — those are session-only.

### Load additions

In GlobalState's `apply_save_data()` (or equivalent), add:

```gdscript
story_arc   = dict.get("story_arc",   {})
story_state = dict.get("story_state", {
    "fired_beats": {}, "arc_flags": {}, "current_arc_index": 0,
    "beats_since_progress": 0, "active_nudge_level": 0
})
```

# Beat dict schema:
#   id          String   unique slug, used as persistence key
#   arc         String   which arc this belongs to (matches story_arc["arcs"][n]["id"])
#   priority    int      lower fires first when multiple beats are ready simultaneously
#   one_shot    bool     if true, never re-fires once in fired_beats
#   cooldown_s  float    minimum seconds since last fire (for repeatable beats), 0 = none
#   check_on    Array    which events evaluate this beat: "system_arrived", "docked",
#                        "kill", "quest_completed", "time_tick"
#   conditions  Array    ALL must pass (see StoryManager._eval_condition)
#   triggers    Array    fire in order when beat passes (see StoryManager._fire_trigger)

## SECTION 3 — StoryManager.gd (the engine)

Full file. This is the autoload singleton. Codex implements this exactly.

```gdscript
# scripts/story/StoryManager.gd
extends Node

const StoryRegistryScript = preload("res://scripts/story/StoryRegistry.gd")

# ── Internal state ────────────────────────────────────────────────────────────
var _beats: Array = []
var _last_fired_time: Dictionary = {}   # beat_id → Time.get_unix_time_from_system()
var _llm_queue: Array = []              # pending Gemma4 requests, processed one at a time
var _llm_busy: bool = false
var _delayed_beat_timers: Array = []    # track active delayed-trigger timers for cleanup

# ── Lifecycle ─────────────────────────────────────────────────────────────────
func _ready() -> void:
    _beats = StoryRegistryScript.get_beats()
    # Restore last-fired times from save so cooldowns survive across sessions
    var saved_times: Dictionary = GlobalState.story_state.get("fired_beats", {})
    for beat_id in saved_times:
        _last_fired_time[beat_id] = int(saved_times[beat_id])

# ── Public event hooks (called by GameRoot, UIManager, etc.) ──────────────────

func on_system_arrived(system_id: String) -> void:
    GlobalState.story_state["beats_since_progress"] = \
        int(GlobalState.story_state.get("beats_since_progress", 0)) + 1
    _evaluate("system_arrived", { "system_id": system_id })
    _check_arc_refresh()
    _check_nudge_escalation()

func on_docked(station_id: String) -> void:
    _evaluate("docked", { "station_id": station_id })

func on_kill(faction: String) -> void:
    _evaluate("kill", { "faction": faction })

func on_quest_completed(quest_type: String, destination_system: String = "") -> void:
    # Clear quest hint if the completed quest matched it
    var hint: Dictionary = GlobalState.story_quest_hint
    if not hint.is_empty():
        if destination_system != "" and destination_system == hint.get("preferred_system", ""):
            GlobalState.story_quest_hint = {}
    _evaluate("quest_completed", { "quest_type": quest_type })

func on_anomaly_activated(anomaly_id: String) -> void:
    _evaluate("anomaly_activated", { "anomaly_id": anomaly_id })

# Call from debug console to force-evaluate all beats regardless of conditions.
# Usage: StoryManager.force_beat("arc1.b1.kaelen_first_intel")
func force_beat(beat_id: String) -> void:
    for beat in _beats:
        if beat.get("id", "") == beat_id:
            _fire_beat(beat)
            return
    push_warning("[StoryManager] force_beat: no beat with id '%s'" % beat_id)

# ── Beat evaluation ───────────────────────────────────────────────────────────

func _evaluate(event: String, ctx: Dictionary) -> void:
    var ready_beats: Array = []
    var now := int(Time.get_unix_time_from_system())
    for beat in _beats:
        if event not in beat.get("check_on", []):
            continue
        if not _beat_passes_cooldown(beat, now):
            continue
        if not _all_conditions_pass(beat.get("conditions", []), ctx):
            continue
        ready_beats.append(beat)
    if ready_beats.is_empty():
        return
    # Sort by priority (lower number = higher priority), fire the top one only
    ready_beats.sort_custom(func(a, b): return a.get("priority", 99) < b.get("priority", 99))
    _fire_beat(ready_beats[0])

func _beat_passes_cooldown(beat: Dictionary, now: int) -> bool:
    var beat_id: String = beat.get("id", "")
    if beat.get("one_shot", false) and beat_id in GlobalState.story_state.get("fired_beats", {}):
        return false
    var cooldown: float = beat.get("cooldown_s", 0.0)
    if cooldown > 0.0 and beat_id in _last_fired_time:
        return (now - _last_fired_time[beat_id]) >= int(cooldown)
    return true

func _fire_beat(beat: Dictionary) -> void:
    var beat_id: String = beat.get("id", "")
    var now := int(Time.get_unix_time_from_system())
    _last_fired_time[beat_id] = now
    GlobalState.story_state["fired_beats"][beat_id] = now
    # Arc progress: reset stall counter
    if beat.get("arc", "ambient") != "ambient":
        GlobalState.story_state["beats_since_progress"] = 0
        GlobalState.story_state["active_nudge_level"] = 0
    for trigger in beat.get("triggers", []):
        _fire_trigger(trigger, beat)

# ── Condition evaluators ──────────────────────────────────────────────────────

func _all_conditions_pass(conditions: Array, ctx: Dictionary) -> bool:
    for c in conditions:
        if not _eval_condition(c, ctx):
            return false
    return true

func _eval_condition(c: Dictionary, ctx: Dictionary) -> bool:
    match c.get("type", ""):
        "flag_set":
            return bool(GlobalState.get(c.get("flag", "")))
        "flag_not_set":
            return not bool(GlobalState.get(c.get("flag", "")))
        "arc_flag":
            return GlobalState.story_state.get("arc_flags", {}).get(c["arc"], "") == c["value"]
        "beat_fired":
            return c["beat_id"] in GlobalState.story_state.get("fired_beats", {})
        "beat_not_fired":
            return c["beat_id"] not in GlobalState.story_state.get("fired_beats", {})
        "systems_visited":
            return GlobalState.kaelen_arrival_systems_seen.size() >= int(c.get("min", 1))
        "kills_total":
            var total := 0
            for b in BountyRegistryScript.shared().get_active_bounties():
                total += int(b.get("kills_credited", 0))
            return total >= int(c.get("min", 1))
        "faction_rep":
            return float(GlobalState.reputations.get(c.get("faction", ""), 0.0)) >= float(c.get("min", 0))
        "random_chance":
            return randf() <= float(c.get("pct", 0.5))
        "system_is_generated":
            # true if current system is NOT home/start system
            return not GlobalState.is_current_system_home()
        "system_not_seen_this_run":
            return GlobalState.current_system_id not in GlobalState.kaelen_arrival_systems_seen
        "anomalies_spawned_this_system":
            var reg = load("res://scripts/AnomalyRegistry.gd")
            return not reg.shared()._last_spawned_flavors.is_empty()
        "has_item":
            return GlobalState.inventory.count(c.get("item_id", "")) > 0
        "in_system":
            return GlobalState.current_system_id == c.get("system_id", "")
        _:
            push_warning("[StoryManager] Unknown condition type: %s" % c.get("type", "?"))
            return false

# NOTE: BountyRegistryScript needs to be preloaded at top of file:
# const BountyRegistryScript = preload("res://scripts/economy/BountyRegistry.gd")
```

## SECTION 4 — StoryManager.gd Trigger Implementations

Add these methods to the same `StoryManager.gd` file, below `_eval_condition`.

```gdscript
# ── Trigger executors ─────────────────────────────────────────────────────────

func _fire_trigger(trigger: Dictionary, beat: Dictionary) -> void:
    match trigger.get("type", ""):
        "kaelen_arrival_line":
            _trigger_kaelen_arrival_line()
        "kaelen_voice_message_delayed":
            _trigger_kaelen_voice_delayed(trigger, beat)
        "anomaly_rumor_chatter":
            _trigger_anomaly_rumor()
        "emit_chatter":
            GlobalState.emit_chatter(
                trigger.get("sender", "SYSTEM"),
                trigger.get("line", ""),
                Color(trigger.get("color", [0.85, 0.85, 0.85]))
            )
        "set_arc_flag":
            GlobalState.story_state["arc_flags"][trigger["arc"]] = trigger["value"]
        "story_quest_hint":
            GlobalState.story_quest_hint = {
                "preferred_system":    trigger.get("preferred_system", ""),
                "preferred_type":      trigger.get("preferred_type", ""),
                "flavor_tag":          trigger.get("flavor_tag", ""),
                "expires_after_docks": int(trigger.get("expires_after_docks", 3)),
            }
        "world_pressure":
            GlobalState.story_world_pressure = {
                "system_id":  trigger.get("system_id", ""),
                "faction":    trigger.get("faction", ""),
                "intensity":  float(trigger.get("intensity", 0.3)),
                "event_type": trigger.get("event_type", "patrol_surge"),
            }
        "request_llm_arc":
            _enqueue_llm_request({ "type": "arc_generation", "beat": beat })
        _:
            push_warning("[StoryManager] Unknown trigger type: %s" % trigger.get("type", "?"))


func _trigger_kaelen_arrival_line() -> void:
    # Mirrors old GameRoot._maybe_emit_kaelen_system_arrival() logic.
    # GameRoot still builds the line (it has system registry access);
    # StoryManager just decides WHEN to allow it.
    # Implementation: call GameRoot directly via scene tree, or move the
    # line-building logic here and pass system_id to a helper.
    #
    # Simplest bridge: emit a signal that GameRoot listens to.
    # Signal: StoryManager.kaelen_arrival_requested(system_id)
    # GameRoot._on_kaelen_arrival_requested(system_id) builds + emits the line.
    kaelen_arrival_requested.emit(GlobalState.current_system_id)


func _trigger_kaelen_voice_delayed(trigger: Dictionary, beat: Dictionary) -> void:
    # Delayed voice messages: queue button on comms panel after random delay.
    var delay := randf_range(
        float(trigger.get("delay_min", 60.0)),
        float(trigger.get("delay_max", 90.0))
    )
    var line := _resolve_beat_line(trigger, beat)
    if line.is_empty():
        return
    var timer := get_tree().create_timer(delay)
    _delayed_beat_timers.append(timer)
    timer.timeout.connect(func():
        _delayed_beat_timers.erase(timer)
        if not is_instance_valid(self):
            return
        # Route through UIManager's public method (see Section 6 — UIManager changes)
        var ui := _find_ui_manager()
        if ui and ui.has_method("queue_kaelen_voice_message"):
            ui.queue_kaelen_voice_message(line)
    )


func _trigger_anomaly_rumor() -> void:
    # Delegate line selection to AnomalyRegistry (it knows the flavor type)
    # then emit via GlobalState chatter.
    var reg = load("res://scripts/AnomalyRegistry.gd")
    var rumor: Dictionary = reg.shared().get_arrival_rumor()
    if rumor.is_empty():
        return
    GlobalState.emit_chatter(
        rumor.get("sender", "Local Traffic"),
        rumor.get("line", ""),
        Color(0.75, 0.75, 0.75)
    )


func _resolve_beat_line(trigger: Dictionary, beat: Dictionary) -> String:
    # Priority: explicit "line" on trigger > line from story_arc beat data > random pool
    var line: String = trigger.get("line", "")
    if not line.is_empty():
        return line
    # Try story_arc for LLM-generated line matching this beat id
    var beat_id: String = beat.get("id", "")
    for arc in GlobalState.story_arc.get("arcs", []):
        for arc_beat in arc.get("beats", []):
            if arc_beat.get("id", "") == beat_id:
                var arc_line: String = arc_beat.get("line", "")
                if not arc_line.is_empty():
                    return arc_line
    # Fall back to random pool
    var pool_key: String = trigger.get("line_pool", "")
    if not pool_key.is_empty():
        var pool: Array = StoryRegistryScript.get_line_pool(pool_key)
        if not pool.is_empty():
            return pool[randi() % pool.size()]
    return ""


func _find_ui_manager() -> Node:
    # UIManager is added as a child of the root scene (GameRoot).
    # Walk the scene tree to find it.
    var root := get_tree().current_scene
    if root and root.has_method("get_ui_manager"):
        return root.get_ui_manager()
    # Fallback: search by class or group
    var candidates := get_tree().get_nodes_in_group("ui_manager")
    if not candidates.is_empty():
        return candidates[0]
    return null


# ── Signals ───────────────────────────────────────────────────────────────────
signal kaelen_arrival_requested(system_id: String)
```

## SECTION 5 — Nudge System + Arc Refresh (StoryManager.gd continued)

Add these methods to StoryManager.gd.

```gdscript
# ── Nudge escalation ──────────────────────────────────────────────────────────

const STALL_THRESHOLD_LIGHT  := 3   # system arrivals before level 1 nudge
const STALL_THRESHOLD_MEDIUM := 5   # before level 2
const STALL_THRESHOLD_HEAVY  := 8   # before level 3 (quest injection)
const STALL_THRESHOLD_WORLD  := 12  # before level 4 (world pressure)

func _check_nudge_escalation() -> void:
    var stall: int = int(GlobalState.story_state.get("beats_since_progress", 0))
    var current_level: int = int(GlobalState.story_state.get("active_nudge_level", 0))
    var next_beat: Dictionary = _get_next_arc_beat()
    if next_beat.is_empty():
        return  # no arc active, nothing to nudge toward

    if stall >= STALL_THRESHOLD_WORLD and current_level < 4:
        _apply_nudge(4, next_beat)
    elif stall >= STALL_THRESHOLD_HEAVY and current_level < 3:
        _apply_nudge(3, next_beat)
    elif stall >= STALL_THRESHOLD_MEDIUM and current_level < 2:
        _apply_nudge(2, next_beat)
    elif stall >= STALL_THRESHOLD_LIGHT and current_level < 1:
        _apply_nudge(1, next_beat)


func _apply_nudge(level: int, next_beat: Dictionary) -> void:
    GlobalState.story_state["active_nudge_level"] = level
    var target_sys: String = _get_arc_target_system()
    match level:
        1:  # hint chatter — passing ship mentions the target system area
            var hints := [
                "Picked up some chatter about activity further out. Worth a look if you've got time.",
                "Someone's moving something through the outer lanes. Noticed it on long-range.",
            ]
            GlobalState.emit_chatter("Local Traffic", hints[randi() % hints.size()], Color(0.7, 0.7, 0.7))
        2:  # Kaelen voice hint
            var ui := _find_ui_manager()
            if ui and ui.has_method("queue_kaelen_voice_message"):
                ui.queue_kaelen_voice_message(
                    "You've been running the same lane for a while. I've got something that might interest you further out."
                )
        3:  # quest injection — bias next quest toward arc target system
            if not target_sys.is_empty():
                GlobalState.story_quest_hint = {
                    "preferred_system":    target_sys,
                    "preferred_type":      next_beat.get("player_need_type", "delivery"),
                    "flavor_tag":          next_beat.get("flavor_tag", ""),
                    "expires_after_docks": 4,
                }
        4:  # world pressure — make current system less comfortable
            GlobalState.story_world_pressure = {
                "system_id":  GlobalState.current_system_id,
                "faction":    _get_pressure_faction(),
                "intensity":  0.5,
                "event_type": "patrol_surge",
            }


func _get_next_arc_beat() -> Dictionary:
    # Returns the first unfired beat in the current arc from story_arc.
    var arc_index: int = int(GlobalState.story_state.get("current_arc_index", 0))
    var arcs: Array = GlobalState.story_arc.get("arcs", [])
    if arc_index >= arcs.size():
        return {}
    var arc: Dictionary = arcs[arc_index]
    var fired: Dictionary = GlobalState.story_state.get("fired_beats", {})
    for beat in arc.get("beats", []):
        if beat.get("id", "") not in fired:
            return beat
    return {}


func _get_arc_target_system() -> String:
    var arc_index: int = int(GlobalState.story_state.get("current_arc_index", 0))
    var arcs: Array = GlobalState.story_arc.get("arcs", [])
    if arc_index < arcs.size():
        return str(arcs[arc_index].get("target_system", ""))
    return ""


func _get_pressure_faction() -> String:
    # Pick the faction with the highest negative rep toward the player,
    # or the arc's designated antagonist faction.
    var arc_index: int = int(GlobalState.story_state.get("current_arc_index", 0))
    var arcs: Array = GlobalState.story_arc.get("arcs", [])
    if arc_index < arcs.size():
        var antagonist: String = str(arcs[arc_index].get("antagonist_faction", ""))
        if not antagonist.is_empty():
            return antagonist
    # Fallback: first faction with negative rep
    for faction in GlobalState.reputations:
        if float(GlobalState.reputations[faction]) < 0.0:
            return faction
    return "zenith"   # last resort default


# ── Arc refresh ───────────────────────────────────────────────────────────────

func _check_arc_refresh() -> void:
    var arc_index: int = int(GlobalState.story_state.get("current_arc_index", 0))
    var arcs: Array = GlobalState.story_arc.get("arcs", [])
    if arc_index >= arcs.size():
        return
    var arc: Dictionary = arcs[arc_index]
    var fired: Dictionary = GlobalState.story_state.get("fired_beats", {})
    var remaining := 0
    for beat in arc.get("beats", []):
        if beat.get("id", "") not in fired:
            remaining += 1
    # Advance arc index if all beats are done
    if remaining == 0:
        GlobalState.story_state["current_arc_index"] = arc_index + 1
        GlobalState.story_state["beats_since_progress"] = 0
        GlobalState.story_state["active_nudge_level"] = 0
    # Pre-generate next arc if we're running low (< 2 beats left)
    elif remaining <= 2:
        var next_index := arc_index + 1
        if next_index >= arcs.size():
            _enqueue_llm_request({ "type": "arc_generation", "arc_number": next_index })
```

## SECTION 6 — LLM Orchestration Queue (StoryManager.gd continued)

```gdscript
# ── LLM orchestration ─────────────────────────────────────────────────────────

func _enqueue_llm_request(request: Dictionary) -> void:
    # Deduplicate: don't queue the same arc_number twice
    for existing in _llm_queue:
        if existing.get("type") == request.get("type") \
        and existing.get("arc_number") == request.get("arc_number"):
            return
    _llm_queue.append(request)
    if not _llm_busy:
        _process_llm_queue()


func _process_llm_queue() -> void:
    if _llm_queue.is_empty() or _llm_busy:
        return
    _llm_busy = true
    var request: Dictionary = _llm_queue.pop_front()
    match request.get("type", ""):
        "arc_generation":
            _request_arc_generation(int(request.get("arc_number", 0)))
        _:
            _llm_busy = false
            _process_llm_queue()


func _request_arc_generation(arc_number: int) -> void:
    # Build context package for Gemma4 and route through LLMInterface.
    # LLMInterface needs a new method: request_story_arc(context, callback).
    # See Section 8 — LLMInterface changes.
    var visited: Array = GlobalState.kaelen_arrival_systems_seen.duplicate()
    var rep_summary := {}
    for faction in GlobalState.reputations:
        rep_summary[faction] = GlobalState.reputations[faction]
    var context := {
        "arc_number":       arc_number,
        "previous_arcs":    _summarize_fired_arc_beats(),
        "player_state": {
            "systems_visited":  visited,
            "kills_total":      _total_kills(),
            "reputations":      rep_summary,
        },
        # galaxy_context is populated by GameRoot/system_registry — see Section 7
        "galaxy_context":   GlobalState.get("story_galaxy_context", {}),
    }
    LLMInterface.request_story_arc(context, func(arc_dict: Dictionary):
        _llm_busy = false
        if arc_dict.is_empty():
            push_warning("[StoryManager] Arc generation returned empty — using fallback")
            _apply_fallback_arc(arc_number)
        else:
            var arcs: Array = GlobalState.story_arc.get("arcs", [])
            while arcs.size() <= arc_number:
                arcs.append({})
            arcs[arc_number] = arc_dict
            GlobalState.story_arc["arcs"] = arcs
        _process_llm_queue()
    )


func _apply_fallback_arc(arc_number: int) -> void:
    # Minimal hardcoded arc so the system never runs completely dry.
    # Codex should expand this with richer fallback content.
    var fallback := {
        "id":               "arc_fallback_%d" % arc_number,
        "name":             "Deepening Tensions",
        "target_system":    "",   # blank = no specific target, nudges won't fire destination
        "antagonist_faction": "zenith",
        "beats": [
            {
                "id":    "fallback.%d.kaelen_check" % arc_number,
                "line":  "Something's shifting out here. I'll be in touch.",
            }
        ]
    }
    var arcs: Array = GlobalState.story_arc.get("arcs", [])
    while arcs.size() <= arc_number:
        arcs.append({})
    arcs[arc_number] = fallback
    GlobalState.story_arc["arcs"] = arcs


func _summarize_fired_arc_beats() -> Array:
    var summary := []
    var fired: Dictionary = GlobalState.story_state.get("fired_beats", {})
    for arc in GlobalState.story_arc.get("arcs", []):
        for beat in arc.get("beats", []):
            if beat.get("id", "") in fired:
                summary.append({ "arc": arc.get("name",""), "beat": beat.get("id",""), "line": beat.get("line","") })
    return summary


func _total_kills() -> int:
    var total := 0
    const BountyRegistryScript = preload("res://scripts/economy/BountyRegistry.gd")
    for b in BountyRegistryScript.shared().get_active_bounties():
        total += int(b.get("kills_credited", 0))
    return total
```

## SECTION 10 — Conflict Analysis

*(This section will be filled in once the background conflict sweep completes.
See the agent results when they arrive. Until then, known risks are:)*

**Known risk 1 — Kaelen arrival line ownership transfer**
GameRoot currently calls `_maybe_emit_kaelen_system_arrival` at the end of the
gate arrival sequence. After the migration, StoryManager owns this decision via
the `kaelen_arrival_requested` signal. If the signal connection fails (timing,
scene not ready), the line will never fire. Guard with a null check and a
fallback direct call.

**Known risk 2 — LLMInterface local call convention**
The `request_story_arc` method needs to use the same local model call path as
existing LLMInterface calls. If the existing path uses coroutines/await rather
than callbacks, adjust the pattern to match. Do not introduce a new HTTP client.

**Known risk 3 — story_quest_hint expiry on rapid re-dock**
If the player docks, leaves, and re-docks quickly, `expires_after_docks` could
decrement twice in one session. The hint uses dock count, not time — acceptable
for now, but note this if hints seem to expire too fast.

**Known risk 4 — StoryManager autoload ordering**
StoryManager._ready() connects to QuestManager's signal. If QuestManager is
listed AFTER StoryManager in the autoload order, QuestManager won't exist yet.
In project.godot, ensure QuestManager appears BEFORE StoryManager in the
[autoload] section.

**Known risk 5 — _find_ui_manager() reliability**
The UIManager is a child of GameRoot, not a group member today.
Section 7b adds `add_to_group("ui_manager")` to fix this. If Codex forgets
that line, all kaelen voice message triggers will silently fail.
Add a push_warning in `_find_ui_manager()` if it returns null.

## APPENDIX — Gemma4 Arc JSON Schema (reference)

Full expected output shape from `request_story_arc`:

```json
{
  "id": "arc_1",
  "name": "Human-readable arc title",
  "target_system": "system_id_from_galaxy",
  "antagonist_faction": "faction.zenith",
  "beats": [
    {
      "id": "arc1.b1.slug",
      "trigger_event": "system_arrived",
      "description": "Internal note — not shown to player",
      "player_need": null,
      "line": "Kaelen's voice message. Short. Her voice.",
      "flavor_tag": "corporate_smuggle"
    }
  ],
  "system_events": [
    {
      "system_id": "frontier_alpha",
      "event": "patrol_surge",
      "intensity": 0.4
    }
  ]
}
```

Valid `trigger_event` values: `system_arrived`, `docked`, `quest_completed`, `kill`, `anomaly_activated`
Valid `event` values in system_events: `patrol_surge`, `faction_pressure`, `trade_disruption`
`player_need` null means no prerequisite. Non-null example: `{ "type": "quest_type", "value": "delivery" }`
